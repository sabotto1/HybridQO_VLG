SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND C.CREATIONDATE<='2014-09-12 19:10:05'  AND PL.CREATIONDATE<='2014-07-15 19:49:54'  AND P.POSTTYPEID=1  AND P.SCORE<=12  AND U.UPVOTES=0  AND V.VOTETYPEID=2
