SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PL.LINKTYPEID=1  AND P.SCORE<=20  AND P.FAVORITECOUNT<=7  AND P.CREATIONDATE<='2014-09-10 16:53:34'  AND V.VOTETYPEID=2
