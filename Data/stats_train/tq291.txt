SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE U.ID = C.USERID 	AND C.USERID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = V.USERID  AND P.POSTTYPEID=1  AND P.COMMENTCOUNT<=11  AND P.CREATIONDATE='2014-02-10 17:01:10'  AND U.VIEWS>=0  AND U.VIEWS<=49  AND U.DOWNVOTES<=0  AND U.CREATIONDATE>='2010-12-03 16:15:57'
