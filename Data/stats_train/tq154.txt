SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.POSTHISTORYTYPEID=2  AND PL.LINKTYPEID=1  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11  AND U.CREATIONDATE>='2010-07-27 02:56:06'
