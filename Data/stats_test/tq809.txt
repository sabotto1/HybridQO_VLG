SELECT  TOP 100 * FROM POSTHISTORY AS PH,          POSTS AS P,  		USERS AS U,  		BADGES AS B  WHERE U.ID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = B.USERID  AND PH.POSTHISTORYTYPEID=5  AND P.POSTTYPEID=2  AND P.SCORE>=-1  AND U.VIEWS<=22  AND U.CREATIONDATE<='2014-08-24 11:49:57'
