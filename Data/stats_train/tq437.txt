SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND P.POSTTYPEID=2  AND P.COMMENTCOUNT>=0  AND P.FAVORITECOUNT>=0  AND U.VIEWS<=49  AND U.UPVOTES>=0  AND U.UPVOTES<=32  AND V.VOTETYPEID=2  AND V.BOUNTYAMOUNT>=0
