SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND B.DATE>='2010-08-26 13:31:17'  AND B.DATE<='2014-08-27 22:03:41'  AND C.SCORE=1  AND U.REPUTATION<=292  AND U.VIEWS=0  AND U.DOWNVOTES<=21  AND U.UPVOTES>=0  AND U.UPVOTES<=10  AND V.VOTETYPEID=2
