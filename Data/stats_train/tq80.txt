SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.CREATIONDATE>='2010-07-20 16:00:09'  AND PH.POSTHISTORYTYPEID=3  AND U.REPUTATION>=1  AND U.REPUTATION<=178  AND U.UPVOTES>=0  AND V.VOTETYPEID=2
