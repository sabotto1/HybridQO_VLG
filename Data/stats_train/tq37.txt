SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.POSTHISTORYTYPEID=2  AND PL.LINKTYPEID=1  AND P.VIEWCOUNT>=0  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11  AND P.CREATIONDATE>='2009-02-02 14:24:58'  AND P.CREATIONDATE<='2014-08-12 19:37:54'  AND U.UPVOTES>=0  AND V.VOTETYPEID=2
