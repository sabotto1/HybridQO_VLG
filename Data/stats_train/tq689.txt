SELECT  TOP 100 * FROM COMMENTS AS C,          POSTHISTORY AS PH,  		BADGES AS B,          USERS AS U  WHERE U.ID = C.USERID 	AND U.ID = PH.USERID 	AND U.ID = B.USERID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=3  AND PH.CREATIONDATE>='2010-12-20 12:30:03'  AND U.REPUTATION=1  AND U.DOWNVOTES<=0  AND U.UPVOTES=0
