SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND PH.POSTHISTORYTYPEID=2  AND P.POSTTYPEID=2  AND P.VIEWCOUNT<=2430  AND P.COMMENTCOUNT>=0  AND U.DOWNVOTES<=17  AND U.UPVOTES>=0  AND U.UPVOTES<=68  AND V.CREATIONDATE>='2010-07-20 00:00:00'  AND V.CREATIONDATE<='2014-09-12 00:00:00'
