SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,           POSTHISTORY AS PH,  		BADGES AS B,          USERS AS U  WHERE U.ID = C.USERID 	AND U.ID = P.LASTEDITORUSERID 	AND U.ID = PH.USERID 	AND U.ID = B.USERID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=2  AND P.POSTTYPEID=2  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=2291  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=7  AND U.UPVOTES<=17
