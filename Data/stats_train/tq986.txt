SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          USERS AS U  WHERE P.ID = PL.RELATEDPOSTID 	AND PL.RELATEDPOSTID = C.POSTID 	AND PL.RELATEDPOSTID = PH.POSTID 	AND U.ID= C.USERID  AND P.COMMENTCOUNT>=0  AND P.CREATIONDATE>='2010-07-20 00:20:51'  AND P.CREATIONDATE<='2014-09-07 03:30:57'  AND U.REPUTATION>=1  AND U.UPVOTES>=0  AND U.CREATIONDATE>='2010-11-15 16:53:42'
