SELECT  TOP 100 * FROM POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE P.ID = PL.RELATEDPOSTID 	AND U.ID = P.OWNERUSERID 	AND U.ID = B.USERID 	AND U.ID = PH.USERID     AND U.ID = V.USERID  AND PH.POSTHISTORYTYPEID=5  AND P.POSTTYPEID=2  AND P.SCORE>=-1  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=15  AND U.REPUTATION>=1  AND U.REPUTATION<=140  AND U.UPVOTES>=0  AND U.CREATIONDATE<='2014-09-09 23:32:58'
