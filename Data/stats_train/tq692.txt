SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND C.SCORE=0  AND C.CREATIONDATE>='2010-08-05 16:48:36'  AND PH.POSTHISTORYTYPEID=3  AND P.SCORE>=0  AND P.SCORE<=31  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11
