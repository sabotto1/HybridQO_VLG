SELECT  TOP 100 * FROM POSTHISTORY AS PH,          POSTS AS P,  		USERS AS U,  		BADGES AS B  WHERE U.ID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = B.USERID  AND PH.POSTHISTORYTYPEID=2  AND P.POSTTYPEID=1  AND P.ANSWERCOUNT=0  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=9  AND U.DOWNVOTES=0  AND U.UPVOTES>=0  AND U.UPVOTES<=9
