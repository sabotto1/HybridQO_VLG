SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=2  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=8483  AND P.ANSWERCOUNT>=0  AND P.COMMENTCOUNT>=0  AND U.REPUTATION<=1443  AND U.DOWNVOTES>=0  AND U.DOWNVOTES<=3
