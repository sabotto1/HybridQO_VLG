SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND PH.CREATIONDATE>='2010-09-15 10:49:45'  AND PH.CREATIONDATE<='2014-09-01 10:48:05'  AND P.SCORE>=0  AND P.SCORE<=11  AND P.ANSWERCOUNT>=0
