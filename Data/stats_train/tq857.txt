SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND C.CREATIONDATE>='2009-04-12 08:13:05'  AND PL.LINKTYPEID=1  AND P.COMMENTCOUNT>=0  AND P.FAVORITECOUNT>=0  AND V.VOTETYPEID=2
