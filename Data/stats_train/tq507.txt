SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTHISTORY AS PH,  		BADGES AS B,          VOTES AS V,          USERS AS U  WHERE U.ID =B.USERID 	AND B.USERID = PH.USERID 	AND PH.USERID = V.USERID 	AND V.USERID = C.USERID  AND C.SCORE=2  AND PH.POSTHISTORYTYPEID=2  AND U.REPUTATION>=1  AND U.REPUTATION<=1159  AND U.UPVOTES>=0  AND V.CREATIONDATE<='2014-09-14 00:00:00'
