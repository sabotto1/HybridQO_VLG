SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND PL.LINKTYPEID=1  AND P.SCORE<=14  AND P.COMMENTCOUNT>=0  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=21  AND V.CREATIONDATE<='2014-09-09 00:00:00'
