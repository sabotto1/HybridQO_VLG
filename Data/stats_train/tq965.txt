SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=1  AND C.CREATIONDATE<='2014-09-11 07:27:31'  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=6283  AND U.VIEWS>=0  AND U.VIEWS<=104  AND U.UPVOTES<=2
