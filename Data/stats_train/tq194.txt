SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.CREATIONDATE<='2014-09-05 09:11:30'  AND PL.CREATIONDATE>='2010-10-01 23:13:10'  AND P.POSTTYPEID=2  AND P.ANSWERCOUNT=0  AND V.VOTETYPEID=2
