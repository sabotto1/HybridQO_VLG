SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.POSTHISTORYTYPEID=5  AND U.VIEWS<=76  AND U.DOWNVOTES>=0  AND U.DOWNVOTES<=0  AND V.CREATIONDATE>='2010-07-21 00:00:00'
