SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND PH.POSTHISTORYTYPEID=5  AND PH.CREATIONDATE<='2014-08-15 08:50:40'  AND V.VOTETYPEID=5
