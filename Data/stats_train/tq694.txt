SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE U.ID = C.USERID 	AND C.USERID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = V.USERID  AND C.SCORE=3  AND PH.CREATIONDATE<='2014-09-09 06:45:04'  AND P.POSTTYPEID=1  AND P.SCORE<=30  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=2290  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=12  AND U.REPUTATION>=1  AND U.REPUTATION<=256  AND U.UPVOTES>=0  AND V.VOTETYPEID=1
