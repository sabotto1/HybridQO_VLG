SELECT  TOP 100 * FROM COMMENTS AS C,          POSTLINKS AS PL,          POSTS AS P,  		USERS AS U,  		BADGES AS B   WHERE P.ID = PL.RELATEDPOSTID 	AND P.ID = C.POSTID 	AND U.ID = B.USERID 	AND U.ID = P.OWNERUSERID  AND B.DATE>='2010-07-20 13:59:12'  AND PL.LINKTYPEID=1  AND PL.CREATIONDATE>='2011-07-29 10:18:28'  AND P.ANSWERCOUNT<=4  AND P.COMMENTCOUNT>=0  AND U.VIEWS>=0  AND U.VIEWS<=37
