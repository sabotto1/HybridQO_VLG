SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.POSTHISTORYTYPEID=2  AND P.POSTTYPEID=1  AND P.VIEWCOUNT<=4063  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=3  AND P.CREATIONDATE>='2010-07-30 17:10:24'  AND U.UPVOTES=0  AND V.CREATIONDATE>='2010-07-21 00:00:00'  AND V.CREATIONDATE<='2014-09-10 00:00:00'
