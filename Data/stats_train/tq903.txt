SELECT  TOP 100 * FROM POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE P.ID = PL.RELATEDPOSTID 	AND U.ID = P.OWNERUSERID 	AND U.ID = B.USERID 	AND U.ID = PH.USERID     AND U.ID = V.USERID  AND PH.POSTHISTORYTYPEID=5  AND P.POSTTYPEID=2  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=2  AND P.CREATIONDATE>='2010-07-20 09:13:29'  AND P.CREATIONDATE<='2014-09-03 15:21:31'  AND U.VIEWS<=111  AND U.UPVOTES>=0  AND V.CREATIONDATE<='2014-09-10 00:00:00'
