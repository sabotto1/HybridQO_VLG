SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=2  AND PL.CREATIONDATE>='2010-10-21 14:19:36'  AND PL.CREATIONDATE<='2014-08-19 15:19:49'  AND P.ANSWERCOUNT<=3
