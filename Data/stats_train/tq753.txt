SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V   WHERE P.OWNERUSERID = C.USERID 	AND P.OWNERUSERID = PH.USERID     AND P.OWNERUSERID = V.USERID  AND C.CREATIONDATE<='2014-09-12 19:47:14'  AND PH.POSTHISTORYTYPEID=3  AND P.SCORE<=53  AND P.COMMENTCOUNT<=16  AND P.FAVORITECOUNT=0  AND V.VOTETYPEID=2  AND V.BOUNTYAMOUNT<=150
