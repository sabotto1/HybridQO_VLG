SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,           POSTHISTORY AS PH,  		BADGES AS B,          USERS AS U  WHERE U.ID = C.USERID 	AND U.ID = P.LASTEDITORUSERID 	AND U.ID = PH.USERID 	AND U.ID = B.USERID  AND B.DATE<='2014-09-11 12:04:01'  AND PH.POSTHISTORYTYPEID=3  AND P.SCORE>=-4  AND P.SCORE<=59  AND P.CREATIONDATE>='2010-07-19 19:39:08'  AND U.REPUTATION<=122  AND U.VIEWS>=0  AND U.VIEWS<=69  AND U.DOWNVOTES<=1  AND U.UPVOTES>=0  AND U.UPVOTES<=7
