SELECT  TOP 100 * FROM COMMENTS AS C,          POSTLINKS AS PL,          POSTS AS P,  		USERS AS U,  		BADGES AS B   WHERE P.ID = PL.RELATEDPOSTID 	AND P.ID = C.POSTID 	AND U.ID = B.USERID 	AND U.ID = P.OWNERUSERID  AND P.SCORE<=18  AND P.ANSWERCOUNT>=0  AND P.COMMENTCOUNT>=0  AND U.DOWNVOTES>=0  AND U.DOWNVOTES<=21  AND U.UPVOTES>=0
