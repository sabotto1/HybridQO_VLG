SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,           POSTHISTORY AS PH,  		BADGES AS B,          USERS AS U  WHERE U.ID = C.USERID 	AND U.ID = P.LASTEDITORUSERID 	AND U.ID = PH.USERID 	AND U.ID = B.USERID  AND B.DATE>='2010-08-03 20:00:23'  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=1  AND P.POSTTYPEID=2  AND P.SCORE>=0  AND P.SCORE<=28  AND P.ANSWERCOUNT>=0  AND P.FAVORITECOUNT>=0  AND U.DOWNVOTES<=0
