SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE P.ID = C.POSTID 	AND P.ID = PL.RELATEDPOSTID     AND P.ID = V.POSTID  	AND U.ID = P.LASTEDITORUSERID     AND U.ID = B.USERID  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=6  AND U.VIEWS>=0  AND U.VIEWS<=34  AND U.UPVOTES<=123  AND V.VOTETYPEID=2  AND V.BOUNTYAMOUNT>=0
