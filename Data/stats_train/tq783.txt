SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND B.DATE>='2010-07-22 12:38:19'  AND B.DATE<='2014-09-10 17:09:59'  AND C.SCORE=0  AND C.CREATIONDATE<='2014-09-12 11:18:24'  AND P.POSTTYPEID=2  AND P.SCORE>=-3  AND P.SCORE<=17  AND P.ANSWERCOUNT<=4  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=17  AND U.REPUTATION>=1  AND U.DOWNVOTES>=0  AND U.UPVOTES>=0
