SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE P.ID = C.POSTID 	AND P.ID = PL.RELATEDPOSTID     AND P.ID = V.POSTID  	AND U.ID = P.LASTEDITORUSERID     AND U.ID = B.USERID  AND P.POSTTYPEID=2  AND P.SCORE>=-4  AND P.SCORE<=30  AND U.DOWNVOTES<=2  AND V.VOTETYPEID=2  AND V.BOUNTYAMOUNT>=0
