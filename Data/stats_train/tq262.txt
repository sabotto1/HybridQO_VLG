SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE U.ID = C.USERID 	AND C.USERID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = V.USERID  AND C.SCORE=2  AND PH.POSTHISTORYTYPEID=2  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=7574  AND P.ANSWERCOUNT=0  AND P.FAVORITECOUNT=0  AND V.CREATIONDATE<='2014-09-13 00:00:00'
