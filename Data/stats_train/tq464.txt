SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE U.ID = C.USERID 	AND C.USERID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = V.USERID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=5  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11  AND P.FAVORITECOUNT>=0  AND U.REPUTATION=148  AND U.VIEWS>=0
