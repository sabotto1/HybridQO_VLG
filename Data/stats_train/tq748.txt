SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND P.FAVORITECOUNT=2
