SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PL.LINKTYPEID=1  AND P.POSTTYPEID=1  AND P.ANSWERCOUNT>=0  AND U.CREATIONDATE>='2010-10-18 13:10:45'  AND U.CREATIONDATE<='2014-08-03 19:24:58'
