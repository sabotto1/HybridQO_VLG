SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=2  AND PL.LINKTYPEID=1  AND P.SCORE>=0  AND P.SCORE<=17  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=8  AND P.CREATIONDATE>='2010-07-20 08:49:13'  AND V.VOTETYPEID=2
