SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE U.ID = C.USERID 	AND C.USERID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = V.USERID  AND C.SCORE=0  AND C.CREATIONDATE<='2014-09-04 03:34:57'  AND PH.POSTHISTORYTYPEID=3  AND P.POSTTYPEID=2  AND P.SCORE=0  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=4  AND U.VIEWS>=0  AND U.VIEWS<=363  AND V.VOTETYPEID=2  AND V.BOUNTYAMOUNT>=0  AND V.BOUNTYAMOUNT<=100
