SELECT  TOP 100 * FROM POSTHISTORY AS PH,          VOTES AS V,  		USERS AS U,  		BADGES AS B  WHERE U.ID = PH.USERID 	AND U.ID = V.USERID 	AND U.ID = B.USERID  AND PH.POSTHISTORYTYPEID=3  AND U.REPUTATION<=1460  AND U.DOWNVOTES<=13  AND U.UPVOTES>=0  AND U.UPVOTES<=24  AND V.CREATIONDATE<='2014-09-10 00:00:00'
