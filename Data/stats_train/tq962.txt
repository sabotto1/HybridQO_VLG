SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND P.COMMENTCOUNT<=9  AND P.FAVORITECOUNT<=14  AND P.CREATIONDATE>='2010-07-19 19:16:27'  AND U.REPUTATION>=1  AND U.REPUTATION<=995
