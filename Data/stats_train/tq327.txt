SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V   WHERE P.OWNERUSERID = C.USERID 	AND P.OWNERUSERID = PH.USERID     AND P.OWNERUSERID = V.USERID  AND C.CREATIONDATE>='2010-07-19 21:54:08'  AND C.CREATIONDATE<='2014-09-12 18:46:24'  AND PH.POSTHISTORYTYPEID=1  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=12  AND P.CREATIONDATE>='2010-07-19 19:47:49'  AND V.VOTETYPEID=1  AND V.CREATIONDATE='2012-07-03 00:00:00'
