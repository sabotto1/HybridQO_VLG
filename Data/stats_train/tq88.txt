SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.CREATIONDATE<='2014-09-12 11:18:24'  AND PH.POSTHISTORYTYPEID=1  AND PH.CREATIONDATE>='2010-09-09 23:09:59'  AND P.VIEWCOUNT>=0  AND P.COMMENTCOUNT>=0
