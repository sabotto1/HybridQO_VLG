SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=5  AND PH.CREATIONDATE<='2014-07-31 16:54:37'  AND P.VIEWCOUNT>=0  AND P.VIEWCOUNT<=6064  AND P.ANSWERCOUNT>=0  AND U.VIEWS>=0  AND U.VIEWS<=18
