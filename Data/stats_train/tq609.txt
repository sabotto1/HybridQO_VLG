SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND C.SCORE=0  AND C.CREATIONDATE<='2014-09-14 01:07:29'  AND PH.POSTHISTORYTYPEID=3  AND PH.CREATIONDATE>='2010-10-22 01:16:57'  AND P.ANSWERCOUNT>=0  AND P.COMMENTCOUNT>=0  AND U.DOWNVOTES<=0  AND U.UPVOTES>=0  AND U.UPVOTES<=3  AND V.VOTETYPEID=2
