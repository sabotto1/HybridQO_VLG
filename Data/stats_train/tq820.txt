SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND PH.POSTHISTORYTYPEID=2  AND P.VIEWCOUNT>=0  AND P.ANSWERCOUNT=0  AND P.CREATIONDATE>='2010-07-23 14:53:50'  AND P.CREATIONDATE<='2014-09-13 19:50:17'  AND V.VOTETYPEID=5
