SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND P.SCORE<=16  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=3  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=22  AND U.UPVOTES<=38
