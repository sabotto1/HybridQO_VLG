SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND P.POSTTYPEID=2  AND P.SCORE<=16  AND P.ANSWERCOUNT<=3  AND P.COMMENTCOUNT<=11
