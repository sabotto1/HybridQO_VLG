SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.POSTHISTORYTYPEID=2  AND PL.CREATIONDATE>='2011-07-20 17:17:56'  AND PL.CREATIONDATE<='2014-04-19 16:14:49'  AND P.COMMENTCOUNT<=20  AND P.FAVORITECOUNT>=0
