SELECT  TOP 100 * FROM COMMENTS AS C,          POSTLINKS AS PL,          POSTS AS P,  		USERS AS U,  		BADGES AS B   WHERE P.ID = PL.RELATEDPOSTID 	AND P.ID = C.POSTID 	AND U.ID = B.USERID 	AND U.ID = P.OWNERUSERID  AND PL.CREATIONDATE<='2014-08-07 11:13:24'  AND P.POSTTYPEID=1  AND P.COMMENTCOUNT>=0
