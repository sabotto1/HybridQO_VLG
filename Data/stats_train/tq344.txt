SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND C.SCORE=1  AND PH.POSTHISTORYTYPEID=5  AND P.POSTTYPEID=1  AND P.VIEWCOUNT>=0  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=6  AND P.COMMENTCOUNT>=0  AND V.VOTETYPEID=2
