SELECT  TOP 100 * FROM POSTHISTORY AS PH,          POSTS AS P,  		USERS AS U,  		BADGES AS B  WHERE U.ID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = B.USERID  AND P.POSTTYPEID=1  AND P.VIEWCOUNT>=0  AND P.COMMENTCOUNT>=0  AND P.FAVORITECOUNT>=0  AND U.VIEWS>=0  AND U.DOWNVOTES>=0  AND U.UPVOTES>=0
