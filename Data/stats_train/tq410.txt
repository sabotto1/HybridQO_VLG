SELECT  TOP 100 * FROM POSTHISTORY AS PH,          POSTS AS P,  		USERS AS U,  		BADGES AS B  WHERE U.ID = P.OWNERUSERID 	AND P.OWNERUSERID = PH.USERID 	AND PH.USERID = B.USERID  AND P.POSTTYPEID=2  AND P.SCORE>=-4  AND P.VIEWCOUNT>=0  AND P.ANSWERCOUNT<=5  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=11  AND U.VIEWS<=28  AND U.UPVOTES>=0
