SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND PH.CREATIONDATE>='2011-05-16 12:12:57'  AND P.COMMENTCOUNT>=0  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=6  AND P.CREATIONDATE>='2010-07-20 19:08:50'  AND P.CREATIONDATE<='2014-08-21 23:01:04'  AND U.REPUTATION>=1  AND U.UPVOTES<=201  AND V.VOTETYPEID=2
