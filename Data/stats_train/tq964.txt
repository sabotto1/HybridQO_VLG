SELECT  TOP 100 * FROM COMMENTS AS C,          POSTLINKS AS PL,          POSTS AS P,  		USERS AS U,  		BADGES AS B   WHERE P.ID = PL.RELATEDPOSTID 	AND P.ID = C.POSTID 	AND U.ID = B.USERID 	AND U.ID = P.OWNERUSERID  AND PL.LINKTYPEID=3  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=13  AND P.FAVORITECOUNT>=0
