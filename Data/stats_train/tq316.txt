SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND C.SCORE=0  AND PH.POSTHISTORYTYPEID=3  AND PL.LINKTYPEID=1  AND PL.CREATIONDATE>='2010-11-12 01:06:02'  AND P.POSTTYPEID=2  AND P.ANSWERCOUNT>=0  AND U.DOWNVOTES<=4
