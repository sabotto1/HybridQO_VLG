SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND PH.POSTHISTORYTYPEID=5  AND PL.LINKTYPEID=1  AND P.POSTTYPEID=2  AND P.ANSWERCOUNT>=0  AND P.CREATIONDATE>='2010-07-20 16:31:26'  AND V.VOTETYPEID=2
