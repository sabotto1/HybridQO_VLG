SELECT  TOP 100 * FROM COMMENTS AS C,          POSTLINKS AS PL,          POSTS AS P,  		USERS AS U,  		BADGES AS B   WHERE P.ID = PL.RELATEDPOSTID 	AND P.ID = C.POSTID 	AND U.ID = B.USERID 	AND U.ID = P.OWNERUSERID  AND C.CREATIONDATE>='2009-04-12 07:48:50'  AND C.CREATIONDATE<='2014-09-13 00:30:45'  AND PL.LINKTYPEID=1  AND P.POSTTYPEID=1  AND P.SCORE>=-2  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=30  AND P.FAVORITECOUNT>=0  AND U.REPUTATION>=1
