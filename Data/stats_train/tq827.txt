SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND PH.POSTHISTORYTYPEID=2  AND PL.CREATIONDATE<='2014-09-07 20:45:27'  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=4  AND V.VOTETYPEID=2
