SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V  WHERE P.ID = C.POSTID 	AND P.ID = PL.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND P.FAVORITECOUNT>=0  AND P.CREATIONDATE<='2014-09-12 00:03:32'  AND V.VOTETYPEID=2  AND V.CREATIONDATE>='2010-07-20 00:00:00'
