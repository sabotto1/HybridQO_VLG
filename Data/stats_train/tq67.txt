SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE U.ID = P.OWNERUSERID     AND U.ID = B.USERID     AND P.ID = C.POSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID  AND C.SCORE=0  AND PH.CREATIONDATE>='2009-12-05 23:10:26'  AND PH.CREATIONDATE<='2014-09-02 02:37:46'  AND P.POSTTYPEID=2  AND P.FAVORITECOUNT>=0  AND P.FAVORITECOUNT<=5  AND U.REPUTATION>=1  AND U.REPUTATION<=284  AND U.VIEWS>=0  AND U.DOWNVOTES>=0  AND U.UPVOTES>=0  AND V.VOTETYPEID=2
