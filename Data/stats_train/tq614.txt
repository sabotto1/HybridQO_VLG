SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTHISTORY AS PH,  		BADGES AS B,          VOTES AS V,          USERS AS U  WHERE U.ID =B.USERID 	AND B.USERID = PH.USERID 	AND PH.USERID = V.USERID 	AND V.USERID = C.USERID  AND C.SCORE=0  AND U.VIEWS<=84  AND U.DOWNVOTES>=0  AND U.DOWNVOTES<=5  AND V.VOTETYPEID=16  AND V.CREATIONDATE>='2010-08-16 00:00:00'  AND V.CREATIONDATE<='2014-09-12 00:00:00'
