SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND B.DATE>='2010-07-19 20:14:08'  AND B.DATE<='2014-08-18 18:02:00'  AND PH.POSTHISTORYTYPEID=2  AND PL.LINKTYPEID=1  AND P.POSTTYPEID=2  AND P.FAVORITECOUNT>=0
