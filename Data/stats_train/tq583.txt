SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTHISTORY AS PH,          VOTES AS V   WHERE P.OWNERUSERID = C.USERID 	AND P.OWNERUSERID = PH.USERID     AND P.OWNERUSERID = V.USERID  AND C.CREATIONDATE>='2009-04-12 07:56:10'  AND C.CREATIONDATE<='2014-09-11 16:34:31'  AND PH.POSTHISTORYTYPEID=5  AND P.POSTTYPEID=1  AND P.ANSWERCOUNT>=0  AND P.ANSWERCOUNT<=5  AND P.COMMENTCOUNT>=0  AND P.COMMENTCOUNT<=11  AND P.FAVORITECOUNT>=0  AND V.CREATIONDATE='2012-04-20 00:00:00'
