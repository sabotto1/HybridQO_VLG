SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,           POSTHISTORY AS PH,  		BADGES AS B,          USERS AS U  WHERE U.ID = C.USERID 	AND U.ID = P.LASTEDITORUSERID 	AND U.ID = PH.USERID 	AND U.ID = B.USERID  AND C.SCORE=0  AND P.POSTTYPEID=1  AND P.VIEWCOUNT>=0  AND P.ANSWERCOUNT>=0  AND P.FAVORITECOUNT=1  AND P.CREATIONDATE<='2014-09-10 16:16:01'  AND U.VIEWS>=0  AND U.DOWNVOTES<=5  AND U.UPVOTES<=3
