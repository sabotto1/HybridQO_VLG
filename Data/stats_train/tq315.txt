SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          VOTES AS V,          BADGES AS B,          USERS AS U  WHERE P.ID = C.POSTID 	AND P.ID = PL.RELATEDPOSTID     AND P.ID = V.POSTID  	AND U.ID = P.LASTEDITORUSERID     AND U.ID = B.USERID  AND U.REPUTATION>=1  AND U.UPVOTES>=0  AND U.UPVOTES<=15
