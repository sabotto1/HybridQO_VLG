SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,          POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          BADGES AS B  WHERE P.ID = C.POSTID     AND P.ID = PL.RELATEDPOSTID     AND P.ID = PH.POSTID     AND P.ID = V.POSTID 	AND B.USERID = C.USERID  AND P.POSTTYPEID=1  AND P.SCORE>=0  AND P.SCORE<=23  AND P.CREATIONDATE>='2010-07-19 21:24:58'  AND P.CREATIONDATE<='2014-09-05 12:21:27'  AND V.VOTETYPEID=2
