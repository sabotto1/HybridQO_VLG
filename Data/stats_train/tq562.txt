SELECT  TOP 100 * FROM COMMENTS AS C,  		POSTS AS P,  		POSTLINKS AS PL,          POSTHISTORY AS PH,          VOTES AS V,          USERS AS U  WHERE P.ID = PL.POSTID 	AND P.ID = C.POSTID 	AND P.ID = PH.POSTID 	AND P.ID = V.POSTID 	AND U.ID = P.LASTEDITORUSERID  AND C.CREATIONDATE>='2010-07-28 09:35:29'  AND P.SCORE<=13  AND P.ANSWERCOUNT<=8  AND U.REPUTATION>=1  AND U.VIEWS>=0  AND U.VIEWS<=34  AND U.DOWNVOTES<=11  AND U.UPVOTES=0  AND U.CREATIONDATE<='2014-08-14 08:28:47'  AND V.CREATIONDATE<='2014-09-09 00:00:00'
